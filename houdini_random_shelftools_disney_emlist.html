Print to a window and the Python Console a list of lights/shaders grouped under each Emission number present. 

import hou

def find_all_emissions():
    # Dictionary to hold emission numbers as keys and lists of matching node paths as values
    emission_dict = {}

    # Loop through the range 1 to 50 (inclusive)
    for emNum in range(1, 51):
        matching_lights = []

        # Loop through all nodes in the scene recursively
        for node in hou.node("/").allSubChildren():
            parm = node.parm("emissionCategory")
            if parm is not None:
                try:
                    if int(parm.eval()) == emNum:
                        matching_lights.append(node)
                except (ValueError, TypeError):
                    continue

        # If we found any matching nodes for this emission number, sort and store them
        if matching_lights:
            sorted_paths = [light.path() for light in sorted(matching_lights, key=lambda x: x.path())]
            emission_dict[emNum] = sorted_paths

    # Check if we found anything across the entire range 1-50
    if not emission_dict:
        hou.ui.displayMessage(
            "No matching items found for emission categories 1 through 50.", 
            severity=hou.severityType.Message,
            title="Emission Search Results"
        )
        return

    # Build the formatted report lines
    report_lines = []
    
    for emNum, paths in sorted(emission_dict.items()):
        report_lines.append(f"\n#  emission {emNum}:")
        for path in paths:
            report_lines.append(path)  # Or add indentation if preferred: f"    {path}"

    # Join everything into a single multi-line string
    message_text = "\n".join(report_lines)

    # 1. Print to the Python Console (making sure it flushes/writes)
    print("\n" + message_text + "\n")

    # Total count of items found for the popup summary
    total_found = sum(len(paths) for paths in emission_dict.values())

    # 2. Display the copyable popup dialog box with the details expanded
    hou.ui.displayMessage(
        f"Found {total_found} item(s) across {len(emission_dict)} emission category/categories (1-50).",
        severity=hou.severityType.Message,
        title="Emission Search Results",
        details=message_text,
        details_expanded=True  # Automatically opens the copyable text drawer
    )

# Run the function
find_all_emissions()
