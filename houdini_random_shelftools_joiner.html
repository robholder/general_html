Shelf Tool - Joiner

Select two or more nodes, and the nodes will be connected down to the last node selected.

Assigned Hotkey: <Alt> + J 
import hou

def connect_to_last_node():
    selected_nodes = hou.selectedNodes()
    
    if len(selected_nodes) < 2:
        hou.ui.displayMessage("Please select at least 2 nodes.")
        return

    # The last selected node is our target destination
    target_node = selected_nodes[-1]
    
    # All nodes before the last one are sources
    source_nodes = selected_nodes[:-1]

    for i, source_node in enumerate(source_nodes):
        try:
            # LOP nodes and others support setInput with an index directly.
            # We wire them sequentially into input 0, 1, 2, etc.
            target_node.setInput(i, source_node)
            print(f"Connected {source_node.name()} to {target_node.name()} at input {i}")
        except Exception as e:
            print(f"Could not connect {source_node.name()} to {target_node.name()}: {e}")

connect_to_last_node()
