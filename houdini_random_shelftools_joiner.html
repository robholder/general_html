Shelf Tool - Joiner

Select two or more nodes, and the nodes will be connected down to the last node selected.

Assigned Hotkey: <Alt> + J 
import hou

def connect_nodes_in_order():
    # Get the selected nodes
    selected_nodes = hou.selectedNodes()
    
    if len(selected_nodes) < 2:
        hou.ui.displayMessage("Please select at least 2 nodes in the order you want them connected.")
        return

    # Loop through the selection and wire each node into the next
    for i in range(len(selected_nodes) - 1):
        source_node = selected_nodes[i]
        target_node = selected_nodes[i + 1]
        
        try:
            # Connect the output of the current node to the first input of the next node
            target_node.setInput(0, source_node)
        except Exception as e:
            print(f"Could not connect {source_node.name()} to {target_node.name()}: {e}")

connect_nodes_in_order()
