#-------------------- Clear Gig, Sequence and Shot Aliases --------------------#
function _clear_nav {
    local target="$1"

    echo "Variables BEFORE clearing:"
    if [ "$target" = "all" ] || [ "$target" = "seq" ]; then
        echo "  PRODUCTION = '${PRODUCTION:-<unset>}'"
        echo "  NAV_SEQ    = '${NAV_SEQ:-<unset>}'"
    fi
    if [ "$target" = "all" ] || [ "$target" = "shot" ]; then
        echo "  NAV_SHOT   = '${NAV_SHOT:-<unset>}'"
    fi

    case "$target" in
        all)
            export PRODUCTION="" NAV_SEQ="" NAV_SHOT=""
            echo "Variables AFTER clearing"
            echo "--> PRODUCTION, NAV_SEQ, and NAV_SHOT cleared."
            ;;
        seq)
            export NAV_SEQ="" NAV_SHOT=""
            echo "Variables AFTER clearing"
            echo "--> NAV_SEQ and NAV_SHOT cleared."
            ;;
        shot)
            export NAV_SHOT=""
            echo "Variables AFTER clearing"
            echo "--> NAV_SHOT cleared."
            ;;
    esac
}

function nogig { _clear_nav "all"; }
function noseq { _clear_nav "seq"; }
function noshot { _clear_nav "shot"; }
#----------------- Clear Gig, Sequence and Shot Aliases - END -----------------#
