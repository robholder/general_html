This was a script I ran to find all the environment variables that might be something to do with keyframes (in a specific facility's environment): 

import os
import hou

print("--- SCANNING OS ENVIRONMENT ---")
for k, v in os.environ.items():
    if any(term in k.lower() for term in ["shot", "seq", "nav", "pipeline", "task"]):
        print(f"{k} = {v}")

print("\n--- SCANNING HOU.SESSION ---")
# Check if your pipeline tools store state in hou.session
for attr in dir(hou.session):
    if not attr.startswith("__"):
        print(f"hou.session.{attr} = {getattr(hou.session, attr)}")

print("\n--- CURRENT HIP FILE ---")

List Methods in Proprietary Library

Here we list methods in a proprietary library (coreservices.prodmanUtils in this case) 

import hou

print("=== CHECKING CORESERVICES FUNCTIONS ===")
try:
    from coreservices import prodmanUtils as pm

    # Print all functions available in the module to see if there's a 'get_current_shot' or similar
    functions = [attr for attr in dir(pm) if callable(getattr(pm, attr))]
    for f in functions:
        if not f.startswith("_"):
            print(f"  - {f}")
except Exception as e:
    print(f"Could not inspect prodmanUtils: {e}")

print("\n=== SCANNING SCENE FOR PIPELINE NODES ===")
# Look for common pipeline or manager nodes at the object level
for node in hou.node("/obj").children():
    if any(
        term in node.type().name().lower()
        for term in ["manager", "shot", "pipeline", "env", "control"]
    ):
        print(f"Found potential manager node: {node.path()} (Type: {node.type().name()})")
